
# Mục đích 

- Sử dụng để tìm đường đi ngắn nhất giữa tất cả các cặp đỉnh của đồ thị
- Áp dụng cho **đồ thị có hướng** và **không có chu trình âm** (**có thể có cạnh âm**)

# Tư tưởng

- Thực hiện quá trình lặp
	- Xét từng đỉnh, với tất cả các đường đi (giữa hai đỉnh bất kỳ), nếu đường đi hiện tại lớn hơn đường đi qua đỉnh đang xét, ta thay lại tành đường đi qua đỉnh này

# Thuật toán và ví dụ

- Mã giả
```cpp
Floyd() {
	// Bước 1: Khởi tạo
	for (int i = 1; i <= n; i++) {
		for (int j = 1; j <= n; j++) {
			d[i][j] = a(i, j);
			if (a(i, j) != \infty) next[i][j] = j;// j là điểm tiếp theo của i->j 
												  // để min (min: 1->2->3
												  // => next[1][3] = 2)
			else next[i][j] = null;
		}
	}

	// Bước 2: Lặp
	for (int k = 1; k <= n; k++) {                // Xét mọi điểm k trung gian
		for (int i = 1; i <= n; i++) {            // Xét mọi điểm bắt đầu i
			for (int j = 1; j <= n; j++) {        // Xét mọi điểm kết thúc j
				if (d[i][j] > d[i][k] + d[k][j]) {// Nếu đi i->k>j ngắn hơn i->j
					d[i][j] = d[i][k] + d[k][j]
					next[i][j] = next[i][k];
				}
			}
		}
	}
}

// Khôi phục đường đi

Reconstruct-Path(u, v) {
	if (next[u][v] == null) 
		<Không tồn tại đường đi từ u -> v>
	else {
		path = {u}; // Bắt đầu từ u
		while (u != v) {
			u = next[u][v];
			path.push_back(u);
		}
	
		return path;
	}
}
```
